-- MySQL 5.7 版本：风险商品出口电子信息提取
-- 对应 Oracle 11g 脚本：存储过程-2-Oracle.txt
--
-- 说明：
-- 1. 已按要求去掉表名前的 schema 前缀（tl_tssh. / tl_admin. / tl_bjts.），即认为原
--    三个 schema 下的表已归并到同一个库中。执行本脚本前请先 USE 到目标库；存储过程
--    内的非限定表名会解析到该过程所属的库。
-- 2. 按要求用嵌套的两层 DECLARE CURSOR 还原 Oracle 的两层 FOR CUR** 循环：
--    外层 CUR_FXGZ 遍历风险规则，内层 CUR_FXSJ 在外层的 BEGIN...END 子块中声明。
--    MySQL 游标的 SELECT 在 OPEN 时才求值其中引用的局部变量，因此内层游标可以直接
--    引用外层刚 FETCH 出来的 V_GZ_* 变量，每轮外层循环 OPEN 一次即得到该规则对应的
--    结果集，无需借助临时表。内层子块自带 NOT FOUND 处理器，作用域内优先于外层的
--    处理器，两层循环的结束标志互不干扰。
-- 3. Oracle 特定函数/语法的转换对照：
--    SYSDATE                    -> NOW()（语句级常量，与 Oracle SYSDATE 语义一致，
--                                  且对基于语句的复制是安全的；SYSDATE() 不是）
--    MAX(SJTB_SJ)-3             -> DATE_SUB(MAX(SJTB_SJ), INTERVAL 3 DAY)
--    SYSDATE-T.FAHQFRQ>365      -> T.FAHQFRQ < DATE_SUB(NOW(), INTERVAL 365 DAY)
--    DATE'2100-12-31'           -> CAST('2100-12-31' AS DATETIME)
--    NVL(a,b)                   -> IFNULL(a,b)
--    DECODE(x,0,0,y)            -> CASE WHEN x=0 THEN 0 ELSE y END
--    SUBSTR                     -> SUBSTRING / LEFT
--    REGEXP_LIKE(c,p)           -> c REGEXP BINARY p
--    a||b                       -> CONCAT_WS('',a,b)
--    TO_CHAR(d,'yyyy-mm-dd')    -> DATE_FORMAT(d,'%Y-%m-%d')
--    ADD_MONTHS(d,12)           -> DATE_ADD(d, INTERVAL 12 MONTH)
--    SEQ_X.NEXTVAL              -> FN_SYS_SEQUENCE_NEXTVAL('SEQ_X')（序列模拟）
--    MERGE ... WHEN NOT MATCHED -> INSERT ... SELECT ... WHERE NOT EXISTS
--    CONTINUE                   -> ITERATE 标签
--    过程末尾的 RETURN;         -> 直接删除（MySQL 的 RETURN 只用于函数）
-- 4. 拼接用 CONCAT_WS('', ...) 而不是 CONCAT()：Oracle 的 || 把 NULL 当空串，而
--    MySQL 的 CONCAT() 只要有一个参数为 NULL 整体就是 NULL，CONCAT_WS 会跳过 NULL。
-- 5. Oracle 把空串视同 NULL，MySQL 不会。规则表中取出的文本列统一套 NULLIF(列,'')，
--    使 "规则项为空则不过滤" 的判断与 Oracle 一致。
-- 6. MySQL 5.7 的 REGEXP 用的仍是 POSIX 正则库：默认按列的排序规则匹配（通常不区分
--    大小写），且按字节而非字符处理。为对齐 Oracle REGEXP_LIKE 默认区分大小写的行为，
--    正则比较统一加 BINARY。另需注意 5.7 不支持 \d、\w 等 Perl 快捷类，规则库中若有
--    这类模式需改写为 [[:digit:]] 等 POSIX 字符类；含中文的模式，字面量匹配没有问题，
--    但字符组 [] 和 . 会按字节生效。
-- 7. MySQL 没有 SEQUENCE，用 sys_sequence 表 + FN_SYS_SEQUENCE_NEXTVAL 函数模拟。
--    开启二进制日志的实例创建该函数需要 SUPER 权限，或设置
--    log_bin_trust_function_creators=1。
-- 8. 保留了原脚本中的 COMMIT。MySQL 默认 autocommit=1 时每条语句已自动提交，这些
--    COMMIT 不产生额外影响；若要完全复现 Oracle 的提交批次，请在调用前 SET
--    autocommit=0。MySQL 的游标是 asensitive 的（结果集已物化），循环中提交不会
--    使游标失效。
-- 9. 各 FETCH 变量的类型按 Oracle 侧的语义给出，实际部署时请与 MySQL 建表 DDL 对齐，
--    避免隐式转换或截断。

SET NAMES utf8;

-- ---------------------------------------------------------------------------
-- Oracle SEQUENCE 的模拟
-- ---------------------------------------------------------------------------
-- SEQ_NAME 上的主键既保证每个序列只有一行，也是并发安全递增的必要条件。
CREATE TABLE IF NOT EXISTS sys_sequence (
    SEQ_NAME  VARCHAR(100) NOT NULL,
    SEQ_VALUE BIGINT       NOT NULL,
    PRIMARY KEY (SEQ_NAME)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- 首次部署时以业务表当前最大 ID 初始化；重复执行不会使序列值倒退。
INSERT INTO sys_sequence (SEQ_NAME, SEQ_VALUE)
SELECT 'SEQ_YJ_DATA_YJXX', IFNULL(MAX(ID), 0) FROM YJ_DATA_YJXX
ON DUPLICATE KEY UPDATE SEQ_VALUE = GREATEST(SEQ_VALUE, VALUES(SEQ_VALUE));

INSERT INTO sys_sequence (SEQ_NAME, SEQ_VALUE)
SELECT 'SEQ_MSG_PUSH_DATA', IFNULL(MAX(ID), 0) FROM MSG_PUSH_DATA
ON DUPLICATE KEY UPDATE SEQ_VALUE = GREATEST(SEQ_VALUE, VALUES(SEQ_VALUE));

INSERT INTO sys_sequence (SEQ_NAME, SEQ_VALUE)
SELECT 'SEQ_FXGL_SZ_SPFXJH', IFNULL(MAX(ID), 0) FROM FXGL_SZ_SPFXJH
ON DUPLICATE KEY UPDATE SEQ_VALUE = GREATEST(SEQ_VALUE, VALUES(SEQ_VALUE));

DELIMITER $$

DROP FUNCTION IF EXISTS FN_SYS_SEQUENCE_NEXTVAL$$
CREATE FUNCTION FN_SYS_SEQUENCE_NEXTVAL(P_SEQ_NAME VARCHAR(100))
RETURNS BIGINT
NOT DETERMINISTIC
MODIFIES SQL DATA
BEGIN
    /*
     * LAST_INSERT_ID(expr) 的值属于当前连接；配合主键冲突更新，可在并发调用时
     * 原子地取得各自的序列值。新序列从 1 开始，已有序列在当前值上加 1。
     */
    INSERT INTO sys_sequence (SEQ_NAME, SEQ_VALUE)
    VALUES (P_SEQ_NAME, LAST_INSERT_ID(1))
    ON DUPLICATE KEY UPDATE SEQ_VALUE = LAST_INSERT_ID(SEQ_VALUE + 1);

    RETURN LAST_INSERT_ID();
END$$

-- ---------------------------------------------------------------------------
-- 风险商品出口电子信息提取
-- 20250331，根据省局李裕军局长意见，对市场采购业务不做风险商品出口短信提醒
-- ---------------------------------------------------------------------------
DROP PROCEDURE IF EXISTS PRO_FXGL_FXSPCK_DZXX$$
CREATE PROCEDURE PRO_FXGL_FXSPCK_DZXX()
BEGIN
    -- Oracle 的 DATE 含时分秒，故用 DATETIME 而不是 MySQL 的 DATE。
    DECLARE LD_BEGINDATE  DATETIME;
    DECLARE V_DONE_FXGZ   TINYINT DEFAULT 0;

    -- 外层游标（风险规则）的 FETCH 变量
    DECLARE V_GZ_MC       VARCHAR(500);
    DECLARE V_GZ_FXMS     VARCHAR(2000);
    DECLARE V_GZ_FW_SWJG  VARCHAR(50);
    DECLARE V_GZ_SPDM     VARCHAR(500);
    DECLARE V_GZ_SPMC     VARCHAR(500);
    DECLARE V_GZ_GGXH     VARCHAR(500);
    DECLARE V_GZ_CKKA     VARCHAR(500);
    DECLARE V_GZ_CKGB     VARCHAR(500);
    DECLARE V_GZ_HYD      VARCHAR(500);
    DECLARE V_GZ_DJQ      DECIMAL(65, 30);
    DECLARE V_GZ_DJZ      DECIMAL(65, 30);
    DECLARE V_GZ_ZJQ      DECIMAL(65, 30);
    DECLARE V_GZ_ZJZ      DECIMAL(65, 30);
    DECLARE V_GZ_YXQZ     DATETIME;
    DECLARE V_GZ_DXTXBZ   VARCHAR(1);

    -- 第一层游标：风险规则
    DECLARE CUR_FXGZ CURSOR FOR
        SELECT NULLIF(GZ_MC, ''),
               NULLIF(GZ_FXMS, ''),
               NULLIF(GZ_FW_SWJG, ''),
               NULLIF(GZ_SPDM, ''),
               NULLIF(GZ_SPMC, ''),
               NULLIF(GZ_GGXH, ''),
               NULLIF(GZ_CKKA, ''),
               NULLIF(GZ_CKGB, ''),
               NULLIF(GZ_HYD, ''),
               GZ_DJQ,
               GZ_DJZ,
               GZ_ZJQ,
               GZ_ZJZ,
               GZ_YXQZ,
               GZ_DXTXBZ
          FROM FXGL_SZ_SPFXCK
         WHERE GZ_FW_DATA IN ('1', '3')   -- 规则适用业务范围：1出口电子信息+3同时适用
           AND QYBJ = 'Y'                 -- 规则启用标记：Y
           -- 规则有效期止：NULL 或大于当前日期
           AND IFNULL(GZ_YXQZ, CAST('2100-12-31' AS DATETIME)) > NOW();

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET V_DONE_FXGZ = 1;

    SELECT DATE_SUB(MAX(T.SJTB_SJ), INTERVAL 3 DAY)
      INTO LD_BEGINDATE
      FROM CKTS_WBSJ_HG_BGD T;

    OPEN CUR_FXGZ;

    READ_FXGZ: LOOP
        FETCH CUR_FXGZ
         INTO V_GZ_MC, V_GZ_FXMS, V_GZ_FW_SWJG, V_GZ_SPDM, V_GZ_SPMC,
              V_GZ_GGXH, V_GZ_CKKA, V_GZ_CKGB, V_GZ_HYD, V_GZ_DJQ,
              V_GZ_DJZ, V_GZ_ZJQ, V_GZ_ZJZ, V_GZ_YXQZ, V_GZ_DXTXBZ;

        IF V_DONE_FXGZ = 1 THEN
            LEAVE READ_FXGZ;
        END IF;

        -- 规则中所有设置为空，无效规则跳过
        IF V_GZ_SPDM IS NULL AND V_GZ_SPMC IS NULL AND V_GZ_GGXH IS NULL
           AND V_GZ_CKKA IS NULL AND V_GZ_CKGB IS NULL AND V_GZ_HYD IS NULL
           AND V_GZ_DJQ IS NULL AND V_GZ_DJZ IS NULL
           AND V_GZ_ZJQ IS NULL AND V_GZ_ZJZ IS NULL THEN
            ITERATE READ_FXGZ;
        END IF;

        /*
         * 第二层游标必须声明在子块的声明区，这里的子块同时起两个作用：
         * 一是让 CUR_FXSJ 拥有独立的 NOT FOUND 处理器，不会把外层的
         * V_DONE_FXGZ 提前置 1；二是每轮外层循环重新进入子块并 OPEN 游标，
         * 使 SELECT 中引用的 V_GZ_* 取到当前这条规则的值。
         */
        BEGIN
            DECLARE V_DONE_FXSJ TINYINT DEFAULT 0;

            -- 内层游标（出口电子信息）的 FETCH 变量
            DECLARE V_FXSJ_DJXH      VARCHAR(50);
            DECLARE V_FXSJ_CKBGDH    VARCHAR(50);
            DECLARE V_FXSJ_MYLAJ     DECIMAL(65, 30);
            DECLARE V_FXSJ_CKNY      VARCHAR(20);
            DECLARE V_FXSJ_CKSP_DM   VARCHAR(50);
            DECLARE V_FXSJ_CKRQ_1    DATETIME;
            DECLARE V_FXSJ_CKBGD18   VARCHAR(18);
            DECLARE V_FXSJ_HGSPMC    VARCHAR(500);
            DECLARE V_FXSJ_NSRDZDAH  VARCHAR(50);
            DECLARE V_FXSJ_SWJG_DM   VARCHAR(50);
            DECLARE V_FXSJ_NSRSBH    VARCHAR(50);
            DECLARE V_FXSJ_NSRMC     VARCHAR(500);

            -- 第二层游标：按当前风险规则扫描出口电子信息
            DECLARE CUR_FXSJ CURSOR FOR
                SELECT T.DJXH,
                       T.CKBGDH,
                       T.MYLAJ,
                       T.CKNY,
                       T.CKSP_DM,
                       T.CKRQ_1,
                       SUBSTRING(T.CKBGDH, 1, 18) AS CKBGD18,
                       T.HGSPMC,
                       S.NSRDZDAH,
                       S.SWJG_DM,
                       IFNULL(NULLIF(S.SHXYNO, ''), S.NSRDJNO) AS NSRSBH,
                       S.NSRMC
                  FROM CKTS_WBSJ_HG_BGD T
                 INNER JOIN GS_DJ_CKTMSDAB S ON S.CPCODE = T.DJXH
                 WHERE T.JGFS_DM <> '1039'
                   AND T.SJTB_SJ >= LD_BEGINDATE
                   -- 规则税务机关为空时，Oracle 的 NULL||'%' 等价于 '%'，故补 IFNULL
                   AND T.TSSWJG_DM_1 LIKE CONCAT(IFNULL(V_GZ_FW_SWJG, ''), '%')
                   AND (V_GZ_SPDM IS NULL
                        OR (LEFT(V_GZ_SPDM, 1) = '!'
                            AND T.CKSP_DM NOT REGEXP BINARY SUBSTRING(V_GZ_SPDM, 2))
                        OR (LEFT(V_GZ_SPDM, 1) <> '!'
                            AND T.CKSP_DM REGEXP BINARY V_GZ_SPDM))
                   AND (V_GZ_SPMC IS NULL OR T.HGSPMC    REGEXP BINARY V_GZ_SPMC)
                   AND (V_GZ_GGXH IS NULL OR T.GGXH      REGEXP BINARY V_GZ_GGXH)
                   AND (V_GZ_CKKA IS NULL OR T.HGGQKA_DM REGEXP BINARY V_GZ_CKKA)
                   AND (V_GZ_CKGB IS NULL OR T.MYGDQSZ_DM REGEXP BINARY V_GZ_CKGB)
                   AND (V_GZ_HYD IS NULL
                        OR (LEFT(V_GZ_HYD, 1) = '!'
                            AND T.HZDWDQ_DM NOT REGEXP BINARY SUBSTRING(V_GZ_HYD, 2))
                        OR (LEFT(V_GZ_HYD, 1) <> '!'
                            AND T.HZDWDQ_DM REGEXP BINARY V_GZ_HYD))
                   AND (V_GZ_DJQ IS NULL
                        OR CASE WHEN T.CKSL = 0 THEN 0 ELSE T.MYLAJ / T.CKSL END > V_GZ_DJQ)
                   AND (V_GZ_DJZ IS NULL
                        OR CASE WHEN T.CKSL = 0 THEN 0 ELSE T.MYLAJ / T.CKSL END < V_GZ_DJZ)
                   AND (V_GZ_ZJQ IS NULL OR T.MYLAJ > V_GZ_ZJQ)
                   AND (V_GZ_ZJZ IS NULL OR T.MYLAJ < V_GZ_ZJZ);

            -- 子块内的处理器在其作用域内优先于外层处理器
            DECLARE CONTINUE HANDLER FOR NOT FOUND SET V_DONE_FXSJ = 1;

            SET V_DONE_FXSJ = 0;
            OPEN CUR_FXSJ;

            READ_FXSJ: LOOP
                FETCH CUR_FXSJ
                 INTO V_FXSJ_DJXH, V_FXSJ_CKBGDH, V_FXSJ_MYLAJ, V_FXSJ_CKNY,
                      V_FXSJ_CKSP_DM, V_FXSJ_CKRQ_1, V_FXSJ_CKBGD18,
                      V_FXSJ_HGSPMC, V_FXSJ_NSRDZDAH, V_FXSJ_SWJG_DM,
                      V_FXSJ_NSRSBH, V_FXSJ_NSRMC;

                IF V_DONE_FXSJ = 1 THEN
                    LEAVE READ_FXSJ;
                END IF;

                -- YJ_DATA_YJXX，态势感知风险管理，所有风险数据往这张表写数据
                INSERT INTO YJ_DATA_YJXX (
                    TBPC, ID, SBID, NSRDZDAH, CJ_DATE, YJCODE, ZBCODE,
                    YJ_OBJECT, YJ_COUNT, YJ_AMT, YJ_RECORD, YJ_MSG,
                    SBYM, SWFLAG, QYFLAG, BMDFLAG
                )
                SELECT 0,
                       FN_SYS_SEQUENCE_NEXTVAL('SEQ_YJ_DATA_YJXX'),
                       0,
                       V_FXSJ_NSRDZDAH,
                       NOW(),
                       '140',
                       '14001',
                       V_FXSJ_CKSP_DM,
                       1,
                       V_FXSJ_MYLAJ,
                       V_FXSJ_CKBGDH,
                       -- 原脚本中“商品名称”取的也是 CKSP_DM，此处照原逻辑保留
                       CONCAT_WS('',
                           '风险商品出口(规则：', IFNULL(V_GZ_MC, V_GZ_FXMS),
                           ')，报关单号(', V_FXSJ_CKBGDH,
                           ')，商品代码(', V_FXSJ_CKSP_DM,
                           ')，商品名称(', V_FXSJ_CKSP_DM,
                           ')，出口美元金额',
                           -- DECIMAL 直接拼接会带出小数尾零，按 Oracle NUMBER 的
                           -- 显示效果先去尾零再拼接（变量标度大于 0，小数点会挡住
                           -- 整数部分的 0，不会误删）
                           TRIM(TRAILING '.' FROM
                                TRIM(TRAILING '0' FROM CAST(V_FXSJ_MYLAJ AS CHAR))),
                           ')'
                       ),
                       V_FXSJ_CKNY,
                       1,
                       0,
                       0
                  FROM DUAL
                 WHERE NOT EXISTS (SELECT 1
                                     FROM YJ_DATA_YJXX A
                                    WHERE A.SBID = 0
                                      AND A.NSRDZDAH = V_FXSJ_NSRDZDAH
                                      AND A.YJCODE = '140'
                                      AND A.ZBCODE = '14001'
                                      AND A.YJ_RECORD = V_FXSJ_CKBGDH)
                   AND NOT EXISTS (SELECT 1
                                     FROM YJ_CS_BMD B
                                     LEFT JOIN YJ_CS_BMD_SUB C ON C.BMDID = B.ID
                                    WHERE B.NSRDZDAH = V_FXSJ_NSRDZDAH
                                      AND B.YJCODE = '140'
                                      AND (B.OBJFLAG = '0' OR C.YJ_OBJECT = V_GZ_MC));

                -- JKGL_GY_FXSM_HGCK，原三三智检风险出口数据，待迁移
                INSERT INTO JKGL_GY_FXSM_HGCK (DJXH, CKBGDH, SMRQ, FXGZ)
                SELECT V_FXSJ_DJXH,
                       V_FXSJ_CKBGDH,
                       NOW(),
                       V_GZ_MC
                  FROM DUAL
                 WHERE NOT EXISTS (SELECT 1
                                     FROM JKGL_GY_FXSM_HGCK A
                                    WHERE A.DJXH = V_FXSJ_DJXH
                                      AND A.CKBGDH = V_FXSJ_CKBGDH)
                   AND NOT EXISTS (SELECT 1
                                     FROM YJ_CS_BMD B
                                     LEFT JOIN YJ_CS_BMD_SUB C ON C.BMDID = B.ID
                                    WHERE B.NSRDZDAH = V_FXSJ_NSRDZDAH
                                      AND B.YJCODE = '140'
                                      AND (B.OBJFLAG = '0' OR C.YJ_OBJECT = V_GZ_MC));

                -- MSG_PUSH_DATA，风险出口数据的短信提醒
                -- 从原 PRO_JKGL_FXCKSP 存储过程迁移到这来，原 biztype 为“风险出口”，
                -- 现改为“风险出口电子信息”
                IF V_GZ_DXTXBZ = 'Y' THEN
                    -- 20240819，插入风险出口的短信提醒信息，
                    -- 风险出口短信提醒按 18 位报关单号汇总
                    INSERT INTO MSG_PUSH_DATA (
                        ID, SWJG_DM, NSRSBH, NSRMC, BIZTYPE, BIZKEY,
                        QDSJ, JZSJ, YWBZ, SJTBSJ, IDLY
                    )
                    SELECT FN_SYS_SEQUENCE_NEXTVAL('SEQ_MSG_PUSH_DATA'),
                           V_FXSJ_SWJG_DM,
                           V_FXSJ_NSRSBH,
                           V_FXSJ_NSRMC,
                           '风险出口电子信息',
                           V_FXSJ_CKBGD18,
                           V_FXSJ_CKRQ_1,
                           NULL,
                           V_GZ_MC,
                           NOW(),
                           '1'
                      FROM DUAL
                     WHERE NOT EXISTS (SELECT 1
                                         FROM MSG_PUSH_DATA A
                                        WHERE A.BIZTYPE = '风险出口电子信息'
                                          AND A.BIZKEY = V_FXSJ_CKBGD18)
                       AND NOT EXISTS (SELECT 1
                                         FROM YJ_CS_BMD B
                                         LEFT JOIN YJ_CS_BMD_SUB C ON C.BMDID = B.ID
                                        WHERE B.NSRDZDAH = V_FXSJ_NSRDZDAH
                                          AND B.YJCODE = '140'
                                          AND (B.OBJFLAG = '0' OR C.YJ_OBJECT = V_GZ_MC));
                END IF;

                COMMIT;
            END LOOP READ_FXSJ;

            CLOSE CUR_FXSJ;
        END;
    END LOOP READ_FXGZ;

    CLOSE CUR_FXGZ;

    -- Oracle MERGE（只有 WHEN NOT MATCHED）等价改写：一年以上未回函
    INSERT INTO FXGL_SZ_SPFXJH (
        ID, TSSWJG_DM, TSSWJG_MC, TBR, TBRQ, GZ_FW_SWJG, GZ_FW_DATA,
        GZ_MC, GZ_FXMS, GZ_GHQYSH, GZ_GHQYMC, GZ_DXTXBZ, QYBJ, FHXXBUUID
    )
    SELECT FN_SYS_SEQUENCE_NEXTVAL('SEQ_FXGL_SZ_SPFXJH'),
           '13300000000',
           '浙江省税务局',
           'SYSTEM',
           NOW(),
           '133',
           '2',
           '一年以上未回函',
           CONCAT_WS('',
               T.FAHDSWJGMC,
               DATE_FORMAT(T.FAHQFRQ, '%Y-%m-%d'),
               '发函，至今尚无明确回函意见'
           ),
           T.GHQYNSRSBH_1,
           T.GHFQYMC_1,
           'N',
           'Y',
           T.FHXXBUUID
      FROM YJ_CS_YCHD_JS T
     WHERE T.FUHLX_DM IS NULL
       AND T.FAHQFRQ < DATE_SUB(NOW(), INTERVAL 365 DAY)
       AND NOT EXISTS (SELECT 1
                         FROM FXGL_SZ_SPFXJH A
                        WHERE A.FHXXBUUID = T.FHXXBUUID
                          AND A.GZ_MC = '一年以上未回函');
    COMMIT;

    -- Oracle MERGE（只有 WHEN NOT MATCHED）等价改写：回函存在异常业务
    INSERT INTO FXGL_SZ_SPFXJH (
        ID, TSSWJG_DM, TSSWJG_MC, TBR, TBRQ, GZ_FW_SWJG, GZ_FW_DATA,
        GZ_MC, GZ_FXMS, GZ_GHQYSH, GZ_GHQYMC, GZ_DXTXBZ, QYBJ,
        FHXXBUUID, GZ_YXQZ
    )
    SELECT FN_SYS_SEQUENCE_NEXTVAL('SEQ_FXGL_SZ_SPFXJH'),
           '13300000000',
           '浙江省税务局',
           'SYSTEM',
           NOW(),
           '133',
           '2',
           '回函存在异常业务',
           CONCAT_WS('',
               T.FAHDSWJGMC,
               DATE_FORMAT(T.FAHQFRQ, '%Y-%m-%d'),
               '发函，对方税务机关回函存在异常业务'
           ),
           T.GHQYNSRSBH_1,
           T.GHFQYMC_1,
           'N',
           'Y',
           T.FHXXBUUID,
           DATE_ADD(T.FUHQFRQ, INTERVAL 12 MONTH)
      FROM YJ_CS_YCHD_JS T
     WHERE T.FUHLX_DM = '2'
       AND NOT EXISTS (SELECT 1
                         FROM FXGL_SZ_SPFXJH A
                        WHERE A.FHXXBUUID = T.FHXXBUUID
                          AND A.GZ_MC = '回函存在异常业务');
    COMMIT;

    /*
     * 原 Oracle 脚本中的第三段 MERGE 处于注释状态，这里保留其 MySQL 等价写法，
     * 同样不启用：
     *
     * INSERT INTO FXGL_SZ_SPFXJH (
     *     ID, TSSWJG_DM, TSSWJG_MC, TBR, TBRQ, GZ_FW_SWJG, GZ_FW_DATA,
     *     GZ_MC, GZ_FXMS, GZ_GHQYSH, GZ_GHQYMC, GZ_DXTXBZ, QYBJ,
     *     FHXXBUUID, GZ_YXQZ
     * )
